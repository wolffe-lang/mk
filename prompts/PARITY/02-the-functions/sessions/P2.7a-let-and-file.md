# P2.7a — `let` and `file` (one session)

GNU make 4.4.1, measured below: `let` expands the names and list first, binds words locally, and expands the body in that copy. The last name keeps the remaining text, including its internal and trailing whitespace. Missing words bind empty; no names still expands the body. Bindings have automatic origin and simple flavor, shadow a command-line variable, and do not expand dollar signs again. Nested bindings leave their caller unchanged.

`file >` overwrites, `>>` appends, and writes return empty. Explicit text gains a newline only if it does not already end in one; omitted text creates a zero-byte file. `<` drops exactly one final newline. A missing read is empty, exit 0: **P2.7 in prompts.md incorrectly calls this an error; the measurement wins.** Everything after the first depth-zero comma is text, including more commas and leading spaces. `<name,extra` is fatal (`file: too many arguments`). Spaces around the filename are trimmed. A recipe can write and read in the same expansion. A recursive variable does no I/O until expanded; an unchosen conditional branch does none. `file` itself does not change `.SHELLSTATUS`.

Use the existing lazy dispatcher for both functions: `let` needs the raw body and `file` must split before expansion can introduce commas. /// Bind already-expanded values without expanding dollar signs again.
/// The last name receives the original tail, including its whitespace.
fn bind_let(mut w: Vars, names: List[str], list: str, line: int) {
    var pos = 0
    var i = 0
    while i < names.len {
        while pos < list.len && let_space(list[pos..pos + 1]) {
            pos += 1
        }
        let start = pos
        if i == names.len - 1 {
            pos = list.len
        } else {
            while pos < list.len && !let_space(list[pos..pos + 1]) {
                pos += 1
            }
        }
        map.set(mut w.table, names[i], make_var(list[start..pos], ":=", line, false, "automatic"))
        i += 1
    }
}

fn let_space(c: str) -> bool {
    c == " " || c == "\t" || c == "\n" || c == "\r"
}

fn file_operation(operation: str, text: str, has_text: bool, line: int) -> str ! {
    Refuse(int, str),
} {
    let op = operation.trim()
    if !op.starts_with(">") && !op.starts_with("<") {
        return Refuse(line, "*** file: invalid file operation: {op}")
    }
    let append = op.starts_with(">>")
    let cut = if append then 2 else 1
    let path = op[cut..].trim()
    if path.len == 0 {
        return Refuse(line, "*** file: missing filename")
    }
    if op.starts_with("<") {
        if has_text {
            return Refuse(line, "*** file: too many arguments")
        }
        let raw = fs.read_text(path) else |e| match e {
            not_found => return "",
            denied => return Refuse(line, "*** open: {path}: Permission denied"),
            io => return Refuse(line, "*** open: {path}: Input/output error"),
            utf8 => return Refuse(line, "*** read: {path}: Invalid UTF-8"),
        }
        let end = if raw.ends_with("\n") then raw.len - 1 else raw.len
        return raw[0..end]
    }
    let data = if has_text && !text.ends_with("\n") then text + "\n" else text
    file_write(path, data, append, line)?
    ""
}

fn file_write(path: str, data: str, append: bool, line: int) -> () ! {Refuse(int, str)} {

import shlex
import shutil
import subprocess
import sys
import tempfile

root = pathlib.Path(sys.argv[1]).resolve()
scratch = pathlib.Path(sys.argv[2])
scratch.mkdir(parents=True, exist_ok=True)
for group in ("functions-let", "functions-file-io"):
    relative = pathlib.Path("tests/corpus") / group
    tracked = subprocess.check_output(
        ["git", "ls-files", "--", str(relative)], cwd=root, text=True
    ).splitlines()
    for row in (root / relative / "rows").read_text().splitlines():
        if not row or row.startswith("#"):
            continue

                dest = pathlib.Path(temp) / str(len(outcomes))
                dest.mkdir()
                for name in tracked:
                    source = root / name
                    target = dest / source.relative_to(root / relative)
                    target.parent.mkdir(parents=True, exist_ok=True)

                    stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
