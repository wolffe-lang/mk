//! functions.lu — gmake's text functions: $(name arg1,arg2,…).
//!
//! A function reference's first word is the name; the rest, after the
//! whitespace that follows the name, is the argument text. Arguments are
//! split on commas at parenthesis/brace depth zero, into at most `max`
//! pieces: the last piece keeps any further commas (gmake's rule for
//! fixed-arity functions).

/// The text functions implemented so far (P2.1b).
fn is_text_function(name: str) -> bool {
    if name == "subst" || name == "strip" || name == "findstring" {
        return true
    }
    if name == "words" || name == "firstword" || name == "lastword" {
        return true
    }
    false
}

/// The argument text of a function reference: everything after the name
/// and the whitespace that follows it.
fn function_args(inside: str) -> str {
    var i = 0
    while i < inside.len && inside[i..i + 1] != " " && inside[i..i + 1] != "\t" {
        i = i + 1
    }
    while i < inside.len && (inside[i..i + 1] == " " || inside[i..i + 1] == "\t") {
        i = i + 1
    }
    inside[i..]
}

/// Split `args` on commas at depth zero into at most `max` pieces.
fn split_args(args: str, max: int) -> List[str] {
    var pieces = List[str]()
    var depth = 0
    var start = 0
    var i = 0
    while i < args.len {
        let c = args[i..i + 1]
        if c == "(" || c == "{{" {
            depth = depth + 1
        } else if c == ")" || c == "}}" {
            depth = depth - 1
        } else if c == "," && depth == 0 && pieces.len < max - 1 {
            (mut pieces).push(args[start..i])
            start = i + 1
        }
        i = i + 1
    }
    (mut pieces).push(args[start..])
    pieces
}

/// The piece at `n`, or "" when there are fewer pieces.
fn arg_at(pieces: List[str], n: int) -> str {
    if n < pieces.len {
        return pieces[n]
    }
    ""
}

/// Replace every occurrence of `from` in `text` with `to` (gmake's subst).
fn replace_all(text: str, from: str, to: str) -> str {
    if from.len == 0 {
        return text
    }
    var result = ""
    var i = 0
    while i < text.len {
        if i + from.len <= text.len && text[i..i + from.len] == from {
            result = result + to
            i = i + from.len
        } else {
            result = result + text[i..i + 1]
            i = i + 1
        }
    }
    result
}

/// Join words with single spaces (gmake's strip).
fn join_words(words: List[str]) -> str {
    var result = ""
    var i = 0
    while i < words.len {
        if i > 0 {
            result = result + " "
        }
        result = result + words[i]
        i = i + 1
    }
    result
}

/// Evaluate a text function. `inside` is the whole expanded reference text
/// (name and arguments); `name` is its first word.
fn call_function(v: Vars, name: str, inside: str, visiting: str, line: int) -> str ! {
    Unterminated(int),
    Cycle(int, str),
    Refuse(int, str),
} {
    let args = function_args(inside)
    if name == "subst" {
        let p = split_args(args, 3)
        return replace_all(arg_at(p, 2), arg_at(p, 0), arg_at(p, 1))
    }
    if name == "strip" {
        return join_words(split_words(args))
    }
    if name == "findstring" {
        let p = split_args(args, 2)
        if arg_at(p, 1).contains(arg_at(p, 0)) {
            return arg_at(p, 0)
        }
        return ""
    }
    let words = split_words(args)
    if name == "words" {
        let n = words.len
        return "{n}"
    }
    if name == "firstword" {
        if words.len > 0 {
            return words[0]
        }
        return ""
    }
    if name == "lastword" {
        if words.len > 0 {
            return words[words.len - 1]
        }
        return ""
    }
    Refuse(line, "functions are not supported")
}
