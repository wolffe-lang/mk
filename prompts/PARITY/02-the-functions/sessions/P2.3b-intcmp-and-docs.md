```
/// An `intcmp` argument: an optional sign and decimal digits (whitespace
/// trimmed), or gmake's fatal line naming the argument as written.
fn int_arg(text: str, which: str, line: int) -> int ! {Refuse(int, str)} {
    let t = text.trim()
    if t.len == 0 {
        return Refuse(line, "*** non-numeric {which} argument to 'intcmp' function: empty value")
    }
    var i = 0
    if t[0..1] == "-" || t[0..1] == "+" {
        i = 1
    }
    if i == t.len {
        return Refuse(line, "*** non-numeric {which} argument to 'intcmp' function: '{t}'")
    }
    var j = i
    while j < t.len {
        let c = t[j..j + 1]
        if c < "0" || c > "9" {
            return Refuse(line, "*** non-numeric {which} argument to 'intcmp' function: '{t}'")
        }
        j = j + 1
    }
    let n = t[i..].to_int() else 0
    if t[0..1] == "-" then 0 - n else n
}
```
