/// `addsuffix`/`addprefix`: the affix on each word.
fn affix_words(suffix: bool, affix: str, words: List[str]) -> str {
    var out = List[str]()
    for w in words {
        (mut out).push(if suffix then w + affix else affix + w)
    }
    join_words(out)
}

/// `join`: the words of both lists concatenated pairwise; the longer
/// list's tail passes through (`$(join a b,c)` is `ac b`).
fn join_pairs(a: List[str], b: List[str]) -> str {
    var out = List[str]()
    var i = 0
    while i < a.len || i < b.len {
        let x = if i < a.len then a[i] else ""
        let y = if i < b.len then b[i] else ""
        (mut out).push(x + y)
        i = i + 1
    }
    join_words(out)
}

/// `abspath`: the word made absolute against the current directory, `.`
/// and empty components dropped, `..` taken (clamped at the root), no
/// trailing slash; the file need not exist (gmake's rule).
fn absolute_path(cwd: str, w: str) -> str {
    let full = if w.starts_with("/") then w else cwd + "/" + w
    var parts = List[str]()
    for piece in full.split("/") {
        if piece == "" || piece == "." {
            continue
        }
        if piece == ".." {
            if parts.len > 0 {
                let n = parts.len - 1
                (mut parts).truncate(n)
            }
            continue
        }
        (mut parts).push(piece)
    }
    if parts.len == 0 {
        return "/"
    }
    var s = ""
    for p in parts {
        s = s + "/" + p
    }
    s
}
```

fn test_function_affix_and_join() {
    let v = new_vars()
    assert((expand(v, "$(addsuffix .c,foo bar)", 1) else "") == "foo.c bar.c", "addsuffix")
    assert((expand(v, "$(addprefix src/,foo bar)", 1) else "") == "src/foo src/bar", "addprefix")
    assert((expand(v, "$(join a b,.c .o)", 1) else "") == "a.c b.o", "join")
    assert((expand(v, "$(join a b,c)", 1) else "") == "ac b", "join: the longer tail passes through")
}

fn test_function_abspath_absolute() {
    let v = new_vars()
    assert((expand(v, "$(abspath /a/./b//c/)", 1) else "") == "/a/b/c", "abspath: dots, doubled and trailing slashes")
    assert((expand(v, "$(abspath /)", 1) else "") == "/", "abspath: the root")
    assert((expand(v, "$(abspath /../../z)", 1) else "") == "/z", "abspath: .. clamps at the root")
}
```
