/// True when a `%` stands among a rule line's target names: before the
/// first `:` at reference depth zero, and itself at depth zero (a `%`
/// inside `$(…)` or `${…}` belongs to a function, not a pattern rule).
fn percent_in_targets(line: str) -> bool {
    var depth = 0
    var i = 0
    while i < line.len {
        let c = line[i..i + 1]
        if c == "(" || c == "{{" {
            depth = depth + 1
        } else if c == ")" || c == "}}" {
            if depth > 0 {
                depth = depth - 1
            }
        } else if c == ":" && depth == 0 {
            return false
        } else if c == "%" && depth == 0 {
            return true
        }
        i = i + 1
    }
    false
}
