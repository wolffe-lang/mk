/// Whether the table holds NAME (the environment does not count).
fn var_defined(v: Vars, name: str) -> bool {
    let _ = map.get(v.table, name) else {
        return false
    }
    true
}

/// The automatic variables' names (`@ < ^ ? * % + |` and their D/F forms).
fn is_automatic_name(name: str) -> bool {
    if name.len == 1 {
        return name == "@" || name == "<" || name == "^" || name == "?" || name == "*" || name == "%" || name == "+" || name == "|"
    }
    name.len == 2 && is_automatic_name(name[0..1]) && (name[1..2] == "D" || name[1..2] == "F")
}
