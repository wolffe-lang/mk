```
/// The file-name functions, one word at a time (gmake's rules): `dir` is
/// everything through the last slash (`./` when there is none); `notdir`
/// is what follows the last slash (an empty piece stays, so gmake prints
/// `[ b]` for `$(notdir src/ b)`); `suffix` is the last `.`-suffix of the
/// last component and a word without one is dropped; `basename` is the
/// word without that suffix.
fn file_name_function(name: str, words: List[str]) -> str {
    var out = List[str]()
    for w in words {
        let slash = w.rfind("/") else -1
        let last = if slash >= 0 then w[slash + 1..] else w
        let dot = last.rfind(".") else -1
        if name == "dir" {
            (mut out).push(if slash >= 0 then w[0..slash + 1] else "./")
        } else if name == "notdir" {
            (mut out).push(last)
        } else if name == "suffix" {
            if dot >= 0 {
                (mut out).push(last[dot..])
            }
        } else if dot >= 0 {
            (mut out).push(w[0..w.len - (last.len - dot)])
        } else {
            (mut out).push(w)
        }
    }
    join_words(out)
}
```

```
fn test_function_dir_notdir() {
    let v = new_vars()
    assert((expand(v, "$(dir src/foo.c hacks)", 1) else "") == "src/ ./", "dir")
    assert((expand(v, "$(dir /foo)", 1) else "") == "/", "dir: root")
    assert((expand(v, "$(notdir src/foo.c hacks)", 1) else "") == "foo.c hacks", "notdir")
    assert((expand(v, "[$(notdir src/ b)]", 1) else "") == "[ b]", "notdir: an empty piece stays")
}

fn test_function_suffix_basename() {
    let v = new_vars()

}
```
