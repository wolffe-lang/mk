```
/// `wildcard`: each pattern's matches in sorted order, each written with
/// the pattern's own directory prefix (`./*.c` gives `./a.c`, `*.c` gives
/// `a.c`); a pattern without `*` or `?` is kept only if it exists.
fn wildcard_words(patterns: List[str]) -> List[str] {
    var out = List[str]()
    for pat in patterns {
        let slash = pat.rfind("/") else -1
        let prefix = if slash >= 0 then pat[0..slash + 1] else ""
        if pat.contains("*") || pat.contains("?") {
            for m in glob(pat) {
                let cut = m.rfind("/") else -1
                (mut out).push(prefix + m[cut + 1..])
            }
        } else if fs.exists(pat) {
            (mut out).push(pat)
        }
    }
    out
}
```
